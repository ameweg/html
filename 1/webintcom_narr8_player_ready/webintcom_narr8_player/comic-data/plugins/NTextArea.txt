Описание (description) текстовой области (NTextArea):
{
    type:0, // тип прокрутки: 0 - вертикальный, 1 - горизонтальный, 2 - в оба направления
    scrollLimit: { // Изменение инерции при выезде текстовой области за границы дива
	x: num, // 0 - инерции за границей не будет, 1- инерция по умолчанию
	y: num  // значение (0,1) снижает "оттяжку" , более 1 увеличивает
    }
    overflow: num// 0 или ничего - hidden, 1 - visible;
    friction: num // инерция, адекватное значение от ~0.7 до ~0.95; если 0 - выключено; по умолчанию 0
    [scrollbars:[описание_скроллбара1, описание_скроллбара2, ...],] // описание скроллбаров
    [fade_top: описание_фейда,] // описание верхнего фейда 
    [fade_bottom: описание_фейда,] // описание нижнего фейда 
    [fade_left: описание_фейда,] // описание левого фейда 
    [fade_right: описание_фейда,] // описание правого фейда 
    text: { // описание прокручиваемой области
        [x: num, ] // начальный отступ от левого края
        [y: num, ] // начальный отступ сверху
        w: num, // ширина
        h: num // высота 
    }
}

описание_скроллбара:
{
    type:0, // тип скроллбара: 0 - вертикальный, 1 - горизонтальный
    
	x: num, //отступ от левого края контента
    y: num, //отступ от верхнего края контента
	w: num, // ширина
	h: num // высота
	
    [arrow_back: описание_элемента_скроллбара,] //стрелка назад
    [underarrow_back: описание_элемента_скроллбара,] //подложка под стрелку назад
    [arrow_forward: описание_элемента_скроллбара,] //стрелка вперед
    [underarrow_forward: описание_элемента_скроллбара,] //подложка вперед
    [line: описание_элемента_скроллбара,] //линия
    [slider: описание_ползунка] //ползунок
}

описание_элемента_скроллбара:
{
    src: 'адрес_изображения_элемента',
    w: num, // ширина элемента
    h: num, // высота элемента
    x: num, //отступ от левого края скроллбара
    y: num, //отступ от верхнего края скроллбара
    [img_w: num, img_h: num] // размеры изображения
    
}

описание_ползунка:
{
    src: 'адрес_изображения_ползунка',
    w: num, // ширина ползунка
    h: num, // высота ползунка
    start: num, // начальная позиция ползунка (верхнего левого угла)
    end: num, // конечная позиция ползунка (верхнего левого угла)
    offset: num // отступ от края таймлайна
}

описание_фейда:
{ 
    src: 'адрес_изображения',
    size: num, // ширина фейда
    [img_w: num, img_h: num] // размеры изображения
}
